mkdir c2
cd c2
sudo ip netns add t2-left
sudo ip netns add t2-right
sudo ip link add t2-l type veth peer name t2-r
sudo ip link set t2-l netns t2-left
sudo ip link set t2-r netns t2-right
sudo ip -n t2-left link set t2-l name lan0
sudo ip -n t2-right link set t2-r name lan0
sudo ip -n t2-left link set lan0 address 02:32:23:02:00:10
sudo ip -n t2-right link set lan0 address 02:32:23:02:00:20
sudo ip -n t2-left link set lo up
sudo ip -n t2-right link set lo up
sudo ip -n t2-left address add 10.32.23.10/24 dev lan0
sudo ip -n t2-right address add 10.32.23.20/24 dev lan0
sudo ip -n t2-left link set lan0 up
sudo ip -n t2-right link set lan0 up
sudo ip -n t2-left -br address | tee left-address.txt
sudo ip -n t2-right -br address | tee right-address.txt
sudo ip -n t2-left neigh flush dev lan0
sudo ip -n t2-right neigh flush dev lan0
sudo ip -n t2-left neigh | tee neighbor-empty.txt
sudo ip netns exec t2-left timeout -s INT 12s tcpdump -U -c 8 -i lan0 -w arp-dynamic.pcap arp or icmp > arp-dynamic.log 2>&1 &
sudo ip netns exec t2-left ping -c 3 -W 1 10.32.23.20 | tee ping-dynamic.txt
sudo ip -n t2-left neigh | tee neighbor-dynamic.txt
sudo ip -n t2-left neigh replace 10.32.23.20 lladdr 02:32:23:02:00:20 nud permanent dev lan0
sudo ip -n t2-right neigh replace 10.32.23.10 lladdr 02:32:23:02:00:10 nud permanent dev lan0
sudo ip -n t2-left neigh | tee neighbor-permanent.txt
sudo ip -n t2-right neigh | tee neighbor-right-permanent.txt
sudo ip netns exec t2-left timeout -s INT 10s tcpdump -U -c 6 -i lan0 -w arp-static.pcap arp or icmp > arp-static.log 2>&1 &
sudo ip netns exec t2-left ping -c 3 -W 1 10.32.23.20 | tee ping-static.txt
sudo ip -n t2-left neigh | tee neighbor-after-static.txt
sudo ip -n t2-left neigh replace 10.32.23.20 lladdr 02:32:23:02:ff:20 nud permanent dev lan0
sudo ip -n t2-left neigh | tee neighbor-wrong.txt
sudo ip netns exec t2-left timeout -s INT 7s tcpdump -U -i lan0 -w wrong-static.pcap arp or icmp > wrong-static.log 2>&1 &
sudo ip -n t2-left neigh
sudo ip netns exec t2-left ping -c 2 -W 1 10.32.23.20 | tee ping-wrong-static.txt
sudo ip -n t2-left neigh | tee neighbor-after-wrong.txt
sudo ip -n t2-left neigh replace 10.32.23.20 lladdr 02:32:23:02:00:20 nud permanent dev lan0
sudo ip netns exec t2-left timeout -s INT 10s tcpdump -U -c 6 -i lan0 -w static-repaired.pcap arp or icmp > static-repaired.log 2>&1 &
sudo ip -n t2-left neigh | tee neighbor-repaired.txt
sudo ip netns exec t2-left ping -c 3 -W 1 10.32.23.20 | tee ping-repaired.txt
history -w history-c2.txt
history -w history-c2-final.txt
